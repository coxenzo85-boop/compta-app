# --- CONNEXION GOOGLE SHEETS (Via Secrets) ---
def get_db_connection():
    # On récupère les infos secrètes stockées dans Streamlit Cloud
    scope = ['https://www.googleapis.com/auth/spreadsheets', 'https://www.googleapis.com/auth/drive']
    credentials_dict = dict(st.secrets["gcp_service_account"])
    creds = Credentials.from_service_account_info(credentials_dict, scopes=scope)
    client = gspread.authorize(creds)
    # Ouvre le fichier Google Sheet par son nom
    sheet = client.open("L3_Compta_Database")
    return sheet

# --- FONCTIONS UTILITAIRES IA & FICHIERS ---
def extract_text_from_file(uploaded_file):
    text = ""
    try:
        if uploaded_file.type == "application/pdf":
            reader = PdfReader(uploaded_file)
            for page in reader.pages:
                text += page.extract_text() + "\n"
        elif uploaded_file.type == "application/vnd.openxmlformats-officedocument.wordprocessingml.document":
            doc = Document(uploaded_file)
            for para in doc.paragraphs:
                text += para.text + "\n"
    except Exception as e:
        return f"Erreur lecture : {e}"
    return text

def get_gemini_response(prompt, context, subject):
    api_key = st.secrets["gemini"]["api_key"] # Récupéré des secrets
    genai.configure(api_key=api_key)
    model = genai.GenerativeModel('gemini-1.5-flash')
    
    system_prompt = f"""
    Tu es un tuteur expert en {subject} (Niveau L3 Compta-Contrôle).
    Réponds en te basant sur le CONTEXTE fourni (le cours).
    Si la réponse n'est pas dans le cours, utilise tes connaissances mais précise-le.
    """
    full_prompt = f"{system_prompt}\n\nCONTEXTE:\n{context}\n\nQUESTION:\n{prompt}"
    try:
        response = model.generate_content(full_prompt)
        return response.text
    except Exception as e:
        return f"Erreur API Gemini : {e}"

# --- NAVIGATION ---
def sidebar_menu():
    st.sidebar.title("☁️ L3 CCA Hub")
    
    # Pomodoro
    st.sidebar.markdown("---")
    st.sidebar.subheader("🍅 Pomodoro")
    col1, col2 = st.sidebar.columns(2)
    if 'pomodoro_active' not in st.session_state:
        st.session_state.pomodoro_active = False
        st.session_state.pomodoro_start = None

    if col1.button("Start 25m"):
        st.session_state.pomodoro_active = True
        st.session_state.pomodoro_start = time.time()
    
    if st.session_state.pomodoro_active:
        elapsed = time.time() - st.session_state.pomodoro_start
        remaining = 25*60 - elapsed
        if remaining > 0:
            mins, secs = divmod(remaining, 60)
            st.sidebar.metric("Reste", f"{int(mins):02}:{int(secs):02}")
        else:
            st.sidebar.success("Pause !")
            st.session_state.pomodoro_active = False

    st.sidebar.markdown("---")
    return st.sidebar.radio("Aller à :", ["Dashboard"] + SUBJECTS)

# --- PAGES ---
def dashboard_page(sh):
    st.title("📊 Dashboard Cloud")
    
    # Récupération données Google Sheets
    try:
        ws_grades = sh.worksheet("Grades")
        grades_data = ws_grades.get_all_records()
        
        ws_tasks = sh.worksheet("Tasks")
        tasks_data = ws_tasks.get_all_records()
    except:
        st.error("Erreur de connexion à Google Sheets. Vérifiez les noms des onglets.")
        return

    # Calculs
    total_points = sum([float(g['Grade'])*float(g['Coefficient']) for g in grades_data if str(g['Grade']).replace('.','',1).isdigit()])
    total_coeff = sum([float(g['Coefficient']) for g in grades_data if str(g['Coefficient']).replace('.','',1).isdigit()])
    gpa = round(total_points / total_coeff, 2) if total_coeff > 0 else 0
    
    urgent_count = len([t for t in tasks_data if t['Status'] == 'À faire'])
    
    col1, col2, col3 = st.columns(3)
    col1.markdown(f"<div class='metric-card'><h3>Moyenne</h3><h1>{gpa}/20</h1></div>", unsafe_allow_html=True)
    col2.markdown(f"<div class='metric-card'><h3>À faire</h3><h1>{urgent_count}</h1></div>", unsafe_allow_html=True)
    col3.markdown(f"<div class='metric-card'><h3>Semaine</h3><h1>S{datetime.now().isocalendar()[1]}</h1></div>", unsafe_allow_html=True)

    st.markdown("---")
    st.subheader("📅 Tâches Globales")
    
    # Affichage tâches
    df_tasks = pd.DataFrame(tasks_data)
    if not df_tasks.empty and 'Status' in df_tasks.columns:
        todo = df_tasks[df_tasks['Status'] == 'À faire']
        if not todo.empty:
            for index, row in todo.iterrows():
                c1, c2, c3 = st.columns([1, 4, 2])
                if c1.button("✅", key=f"d_{row['ID']}"):
                    # Update Google Sheet
                    cell = ws_tasks.find(row['ID'])
                    ws_tasks.update_cell(cell.row, 4, "Fait") # 4eme colonne = Status
                    st.success("Tâche terminée !")
                    time.sleep(1)
                    st.rerun()
                c2.write(f"**{row['Subject']}**: {row['Task']}")
                c3.caption(f"Pour le : {row['Due_Date']}")
        else:
            st.info("Rien à faire !")
    else:
        st.info("Aucune tâche enregistrée.")

def subject_page(sh, subject):
    st.title(f"📘 {subject}")
    
    tab1, tab2, tab3 = st.tabs(["💬 Tuteur IA (RAG)", "📝 Notes", "✅ Tâches"])
    
    # --- IA ---
    with tab1:
        st.info("ℹ️ Cloud Mode : Glissez vos cours ici à chaque session pour que l'IA puisse les lire.")
        uploaded_files = st.file_uploader("Documents du cours", type=['pdf', 'docx'], accept_multiple_files=True)
        
        context_text = ""
        if uploaded_files:
            for up_file in uploaded_files:
                context_text += extract_text_from_file(up_file) + "\n"
            st.success(f"{len(uploaded_files)} documents chargés en mémoire.")

        # Chat
        if "messages" not in st.session_state:
            st.session_state.messages = {}
        if subject not in st.session_state.messages:
            st.session_state.messages[subject] = []

        for msg in st.session_state.messages[subject]:
            with st.chat_message(msg["role"]):
                st.markdown(msg["content"])

        if prompt := st.chat_input("Pose une question sur tes documents..."):
            if not context_text:
                st.warning("Veuillez charger un document d'abord.")
            else:
                st.session_state.messages[subject].append({"role": "user", "content": prompt})
                with st.chat_message("user"):
                    st.markdown(prompt)
                
                with st.chat_message("assistant"):
                    with st.spinner("L'expert réfléchit..."):
                        response = get_gemini_response(prompt, context_text, subject)
                        st.markdown(response)
                st.session_state.messages[subject].append({"role": "assistant", "content": response})

    # --- NOTES ---
    with tab2:
        ws_grades = sh.worksheet("Grades")
        with st.form("add_grade"):
            c1, c2, c3 = st.columns(3)
            grade = c1.number_input("Note", 0.0, 20.0)
            coef = c2.number_input("Coef", 0.0, 10.0, value=1.0)
            g_type = c3.selectbox("Type", ["CC", "Partiel", "Examen"])
            if st.form_submit_button("Ajouter Note"):
                new_id = str(uuid.uuid4())[:8]
                ws_grades.append_row([new_id, subject, grade, coef, g_type])
                st.success("Sauvegardé sur Drive !")
                time.sleep(1)
                st.rerun()
        
        # Affichage
        all_grades = ws_grades.get_all_records()
        df = pd.DataFrame(all_grades)
        if not df.empty:
            df_sub = df[df['Subject'] == subject]
            if not df_sub.empty:
                st.dataframe(df_sub[["Note" if "Note" in df_sub else "Grade", "Coefficient", "Type"]])
            else:
                st.info("Pas de notes.")

    # --- TÂCHES ---
    with tab3:
        ws_tasks = sh.worksheet("Tasks")
        with st.form("add_task"):
            col_t, col_d = st.columns([3, 1])
            task_txt = col_t.text_input("Tâche")
            d_date = col_d.date_input("Date")
            if st.form_submit_button("Ajouter"):
                new_id = str(uuid.uuid4())[:8]
                ws_tasks.append_row([new_id, subject, task_txt, "À faire", str(d_date)])
                st.rerun()

        all_tasks = ws_tasks.get_all_records()
        df_t = pd.DataFrame(all_tasks)
        if not df_t.empty:
            df_sub = df_t[(df_t['Subject'] == subject) & (df_t['Status'] == 'À faire')]
            for index, row in df_sub.iterrows():
                st.write(f"▫️ {row['Task']} ({row['Due_Date']})")
